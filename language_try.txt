fn start(): ##ON start it runs this function
    set_input_setting()
    draw_player()

fn set_input_setting():
    input_window.show_window() ##shows the given window
    input_window.title = "Input Window" ##sets window title

    ##up
    up_window_input_button.init() ##instantiates the editable button
    up_window_input_button.text = "Press a button to set the 'up' direction" ##sets button title to the given title
    input.up = await input.wait_for_next_press() ##waits until it gets a random input (except MODE)
    up_window_input_button.input = input.up ##sets the given button input to the random button that we earlyer set

    ##down
    down_window_input_button.init()
    down_window_input_button.text = "Press a button to set the 'down' direction"
    input.down = await input.wait_for_next_press() 
    down_window_input_button.input = input.down

    ##right
    right_window_input_button.init()
    right_window_input_button.text = "Press a button to set the 'right' direction"
    input.right = await input.wait_for_next_press() 
    right_window_input_button.input = input.right

    ##left
    left_window_input_button.init()
    left_window_input_button.text = "Press a button to set the 'left' direction"
    input.left = await input.wait_for_next_press() 
    left_window_input_button.input = input.left

fn draw_player():
    player.init() ##instantiates the player
    player.position = Vector2(1 , 1) ##sets player position (origin is at 0 , 0)

fn process(delta): ##runs on every frame after start
    if input.just_pressed == input.up:
        player.position.y += 1
    if input.just_pressed == input.down:
        player.position.y += -1 ##or  -=1
    if input.just_pressed == input.right:
        player.position.x += 1
    if input.just_pressed == input.left:
        player.position.x += -1 ##or  -=1
    else:
        return

Resiris language elements and defined words
########################################################################

## The basic keywords of the language

include:
    Includes another module or file in the program.
    Example: <include:"Resiris.module">

v:
    Defines a variable.
    Example: v number1
c:
    Defines a constans
    Cant change its value or valuetype
fn:
    Defines a function. A function is a named block of code
    that can be called later.
    Example: fn start():

start():
    A function that runs once when the program starts.

process(delta):
    A function that runs on every frame while the game is running.
    delta stores the time passed since the previous frame.

await:
    Waits for an event or result before continuing the function.
    Example: input.up = await input.wait_for_next_press()

if:
    Checks a condition. Its block runs only when the condition is true.

else:
    The counterpart of if. It runs when the if condition is false.

match:
    Compares a value with multiple possible options.
    The code in the matching branch is executed.

pass:
    Performs no operation. It is used as a placeholder for an empty block.
    It does not stop the function and does not ignore code after the block.

## Defined Resiris and ResirisGameEngine elements

Resiris.module:
    The main Resiris module, containing the basic system elements.

ResirisGameEngine:
    The module containing the game-related systems.

player:
    The player object. After initialization, its position can be set
    with the position property.

input:
    The input system object. just_pressed represents the button
    that was just pressed.

window:
    A window object that can be displayed and configured.

editable_button:
    A button to which the user can assign an input.

Vector2:
    A value that stores two-dimensional coordinates, such as Vector2(1, 1).

init():
    Initializes or creates an object.

show_window():
    Displays the window.

wait_for_next_press():
    Waits for the next user button press and returns the pressed button.

position:
    A property that stores an object's position.

title and text:
    Properties that store the displayed text of a window or button.

## Indentation determines which block each line of code belongs to.
## The language structure is based on Godot/GDScript, but Resiris has
## its own keywords and objects that must be defined separately.
